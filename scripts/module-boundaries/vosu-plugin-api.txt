# Dependencies between vosu-plugin-api's own top-level src/ modules,
# enforced by scripts/check-module-boundaries.py.
#
# The checker's crate::X regex only resolves X when it is itself a module
# name (crate::cbor::..., crate::{cbor, flags}); it cannot see an edge
# created by importing a symbol re-exported at the crate root
# (crate::vosu_status, crate::{vosu_plan_id, ...}). Several real edges
# below are only visible that way; they are listed deliberately so the
# graph stays accurate even where the checker itself cannot derive them.
manifest -> limits
manifest -> capability
catalog -> manifest
cbor -> validate
# B0 (AF.12): the artifact name rule shares the header's name bound.
cbor -> limits
message -> cbor
message -> flags
message -> flat
message -> opcode
message -> primitives
message -> status
message -> validate
message -> value
sdk -> cbor
sdk -> envelope
# S0 (AF.9) moved `PlanMemory` and its flat read-plan imports to
# vosu-plugin-internal; the SDK glue keeps only the ABI-limit, message,
# opcode, status and vtable edges.
sdk -> limits
sdk -> message
sdk -> opcode
sdk -> status
sdk -> vtable
vtable -> envelope
validate -> flags
validate -> flat
validate -> limits
validate -> opcode
validate -> value

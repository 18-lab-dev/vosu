# Dependencies between vosu-plugin-api's own top-level src/ modules,
# enforced by scripts/check-module-boundaries.py.
#
# The checker's crate::X regex only resolves X when it is itself a module
# name (crate::cbor::..., crate::{cbor, flags}); it cannot see an edge
# created by importing a symbol re-exported at the crate root
# (crate::vosu_status, crate::{vosu_plan_id, ...}). Several real edges
# below are only visible that way; they are listed deliberately so the
# graph stays accurate even where the checker itself cannot derive them.
manifest -> limits
manifest -> capability
cbor -> validate
message -> cbor
message -> flags
message -> flat
message -> opcode
message -> primitives
message -> status
message -> validate
message -> value
sdk -> cbor
sdk -> envelope
# F4: `sdk::plan_memory`'s zero-alloc `read_flat` imports the raw flat
# read-plan structs/encoder and the `vosu_status`-returning validate
# helpers directly (re-exported at the crate root), the same ABI-struct
# layer `message` itself builds on — not a new ownership direction, just
# one the checker's regex cannot see through a crate-root re-export
# (comment above).
sdk -> flags
sdk -> flat
sdk -> validate
sdk -> limits
sdk -> message
sdk -> opcode
sdk -> primitives
